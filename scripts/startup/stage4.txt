# Стартовый скрипт этапа 4: ls, cd, clear, uname.
# Запуск: run.sh --vfs build/vfs/deep.zip --script scripts/startup/stage4.txt

# --- clear: очистка экрана (в начале, чтобы не стереть остальной вывод)
clear
vfs-info

# --- ls: все режимы
ls                          # текущий каталог (корень VFS)
ls -l                       # длинный формат: тип, размер, имя
ls docs                     # содержимое каталога
ls docs/notes/archive/2025  # четвёртый уровень вложенности
ls README.txt               # файл
ls -l images                # двоичный файл
ls empty                    # пустой каталог
ls README.txt docs empty    # несколько путей
ls /docs/notes ~/src        # абсолютный путь и путь от ~

# --- cd: все режимы, приглашение показывает текущий каталог
cd docs                     # относительный путь
ls
cd notes/archive            # несколько уровней
cd ..                       # родительский каталог
cd /src/lib                 # абсолютный путь
ls -l
cd -                        # предыдущий каталог
cd .                        # текущий каталог
cd ~/docs                   # путь от домашнего каталога
cd                          # домашний каталог (корень VFS)

# --- uname: данные реальной ОС
uname                       # имя ядра
uname -n                    # имя узла сети
uname -sr                   # объединённые опции
uname -s -m                 # отдельные опции
uname -a                    # все сведения

# --- обработка ошибок
ls nope                     # нет такого пути
ls nope docs                # ошибка не прерывает вывод остальных путей
ls -z                       # неизвестная опция
cd nope                     # нет такого каталога
cd README.txt               # файл вместо каталога
cd docs src                 # лишний аргумент
clear now                   # лишний аргумент
uname -x                    # неизвестная опция
uname foo                   # лишний операнд

exit 0
